```

    --Namespace qce/block_storage \
    --MetricName disk_usage \
    --Period 300 \
    --StartTime 2020-06-09T17:31:13+08:00 \
    --EndTime 2020-06-09T18:31:13+08:00 \
    --Dimensions.0.Name diskId \
    --Dimensions '[{"InstanceId": "lhins-k31c771o"}]'
```
